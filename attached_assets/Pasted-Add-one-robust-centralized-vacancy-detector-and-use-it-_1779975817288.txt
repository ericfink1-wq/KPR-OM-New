Add one robust, centralized vacancy detector and use it everywhere, replacing the weak exact-match "vacant" checks that are scattered across the app. A vacant/empty space currently only registers if its name is exactly "Vacant" — so variants like "Available", "Vacant Suite", "Vacant - Suite 200", "Spec Space", "White Box", "Vacancy", or an empty name slip through and get wrongly treated as real tenants. Fix this.

PART 1 — In src/lib/utils.ts, add and export a function isVacant:

export function isVacant(name: unknown): boolean {
  const s = String(name ?? "").trim().toLowerCase().replace(/\s+/g, " ");
  if (!s || s === "-" || s === "–" || s === "—" || s === "n/a" || s === "na") return true;
  const c = s.replace(/^[\s\-–—•·*"']+/, ""); // strip leading bullets/dashes/quotes
  if (/^vacan/.test(c)) return true;                 // vacant, vacancy, vacancies
  if (/^availab/.test(c)) return true;               // available, availability
  if (/^spec\s+(suite|space|unit)/.test(c)) return true; // spec suite/space/unit
  if (/white\s*box/.test(c)) return true;            // white box / whitebox
  return false;
}
