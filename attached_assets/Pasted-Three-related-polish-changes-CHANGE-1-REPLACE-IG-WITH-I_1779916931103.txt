(a) In artifacts/om-database/src/lib/tenantCredit.ts, change the isInvestmentGrade signature to accept an optional creditRating parameter, and check it before/after the name match:

    export function isInvestmentGrade(tenantName: string, creditRating?: string | null): boolean {
      // Check creditRating field first — explicit overrides the static list
      if (creditRating && typeof creditRating === "string") {
        const r = creditRating.toLowerCase();
        // Match "investment grade", "ig-rated", "BBB", "A-", "A1", "AA", "AAA" etc.
        if (
          r.includes("investment grade") ||
          /\b(aaa|aa[+-]?|a[+-]?|bbb[+-]?)\b/i.test(creditRating)
        ) return true;
      }
      // Existing name-based check (unchanged logic)
      if (!tenantName) return false;
      const cleaned = tenantName
        .toLowerCase()
        .replace(/\(.*?\)/g, "")
        .replace(/#\s*[\w-]+/g, "")
        .replace(/\b(corporation|corp|inc|llc|ltd|co|stores?)\b\.?/g, "")
        .replace(/[^\w\s']/g, " ")
        .replace(/\s+/g, " ")
        .trim();
      for (const brand of IG_TENANTS) {
        const b = brand.toLowerCase();
        if (cleaned === b) return true;
        const padded = " " + cleaned + " ";
        if (padded.includes(" " + b + " ")) return true;
        if (!b.includes(" ") && b.length >= 4 && cleaned.includes(b)) return true;
      }
      return false;
    }

CHANGE 3 — FAST + CLICKABLE TOOLTIP FOR THE ⚑ FLAG IN TENANT ROSTER.

Currently artifacts/om-database/src/components/TenantRoster.tsx line 110 uses the browser's native `title` attribute on the ⚑ flag emoji:

  {t.assumptionNote && <span title={t.assumptionNote} style={{ marginLeft:6, color:"#b45309", cursor:"help" }}>⚑</span>}

Two problems with native `title`: the browser's hover delay is ~700ms (slow), and there's no way to view it on touch devices (no hover). Replace with a small inline tooltip component that:
  • Shows on mouse hover with 150ms delay (much faster than browser native)
  • Shows on click/tap (works on mobile, also lets desktop users pin it open)
  • Dismisses on mouse leave (after hover-open) OR on click outside (after click-open)

(a) At the top of TenantRoster.tsx, near the existing imports, add the necessary React imports if not present:
    import { useState, useRef, useEffect } from "react";

(b) Add a small FlagTip component inside the TenantRoster.tsx file, above the default export. Don't put it in a separate file — keep it local since it's only used here.

    function FlagTip({ content, children, color = "#b45309" }: { content: string; children: React.ReactNode; color?: string }) {
      const [open, setOpen] = useState(false);
      const showTimer = useRef<number | null>(null);
      const ref = useRef<HTMLSpanElement>(null);

      const showSoon = () => {
        if (showTimer.current) window.clearTimeout(showTimer.current);
        showTimer.current = window.setTimeout(() => setOpen(true), 150);
      };
      const cancelHover = () => {
        if (showTimer.current) {
          window.clearTimeout(showTimer.current);
          showTimer.current = null;
        }
        setOpen(false);
      };

      // Close on outside click/tap (only when opened by click)
      useEffect(() => {
        if (!open) return;
        const handler = (e: MouseEvent | TouchEvent) => {
          if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false);
        };
        document.addEventListener("mousedown", handler);
        document.addEventListener("touchstart", handler);
        return () => {
          document.removeEventListener("mousedown", handler);
          document.removeEventListener("touchstart", handler);
        };
      }, [open]);

      return (
        <span ref={ref} style={{ position: "relative", display: "inline-block", marginLeft: 6 }}>
          <span
            onMouseEnter={showSoon}
            onMouseLeave={cancelHover}
            onClick={(e) => { e.stopPropagation(); setOpen(o => !o); }}
            style={{ color, cursor: "pointer", userSelect: "none" }}
            aria-label={content}
            role="button"
            tabIndex={0}
          >
            {children}
          </span>
          {open && (
            <span
              onClick={(e) => e.stopPropagation()}
              style={{
                position: "absolute",
                top: "calc(100% + 6px)",
                left: 0,
                background: "#26281f",
                color: "#f6f2ea",
                padding: "9px 13px",
                borderRadius: 7,
                fontSize: 12,
                lineHeight: 1.5,
                whiteSpace: "normal",
                width: "max-content",
                maxWidth: 320,
                zIndex: 10000,
                boxShadow: "0 8px 24px rgba(0,0,0,0.22)",
              }}
            >
              {content}
            </span>
          )}
        </span>
      );
    }
